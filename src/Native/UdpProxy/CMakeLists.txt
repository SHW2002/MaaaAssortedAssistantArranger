cmake_minimum_required(VERSION 3.23)
project(MaaaUdpProxy LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "MaaaUdpProxy requires MSVC and native Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Static CRT: no separate VC++ or .NET Framework installation for this component.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(MaaaUdpProxy main.cpp third_party/proxifyre/ndisapi/ndisapi.cpp)
target_include_directories(MaaaUdpProxy PRIVATE third_party/gsl/include)
target_compile_definitions(MaaaUdpProxy PRIVATE _LIB _CRT_RAND_S UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN=)
target_compile_options(MaaaUdpProxy PRIVATE /W3 /EHsc /utf-8 /bigobj /permissive-)
target_link_libraries(MaaaUdpProxy PRIVATE iphlpapi ws2_32 advapi32 dnsapi secur32 crypt32)
add_executable(MaaaUdpPolicyTests tests/policy_tests.cpp)
target_include_directories(MaaaUdpPolicyTests PRIVATE third_party/gsl/include)
target_compile_definitions(MaaaUdpPolicyTests PRIVATE _LIB _CRT_RAND_S NOMINMAX UNICODE _UNICODE WIN32_LEAN_AND_MEAN=)
target_compile_options(MaaaUdpPolicyTests PRIVATE /EHsc /utf-8)
target_link_libraries(MaaaUdpPolicyTests PRIVATE iphlpapi ws2_32 advapi32 dnsapi secur32 crypt32)
enable_testing()
add_test(NAME udp-policy COMMAND MaaaUdpPolicyTests)
